     4. Statistical_Testing_Decision_Tree.md (QUICK LOOKUP)

     Purpose: Visual flowchart to decide which statistical test to use

     Contents:
     START: What are you comparing?
     │
     ├─ Comparing PROPORTIONS (%, rates, success/failure)?
     │  │
     │  ├─ Two groups (A vs B)?
     │  │  └─ Use: Chi-Squared Test OR Proportion Test
     │  │     R Code: chisq.test() or prop.test()
     │  │     Example: Success rate Group A (25%) vs Group B (30%)
     │  │
     │  └─ One group vs expected value?
     │     └─ Use: Binomial Test or Chi-Squared Goodness of Fit
     │        R Code: binom.test()
     │
     ├─ Comparing MEANS (averages, continuous values)?
     │  │
     │  ├─ Two groups (A vs B)?
     │  │  └─ Use: Independent t-test
     │  │     R Code: t.test(groupA, groupB)
     │  │     Example: Avg session length Group A (20 min) vs Group B (25
     min)
     │  │
     │  ├─ Same group, before/after?
     │  │  └─ Use: Paired t-test
     │  │     R Code: t.test(before, after, paired=TRUE)
     │  │
     │  └─ More than 2 groups?
     │     └─ Use: ANOVA
     │        R Code: aov()
     │
     └─ Comparing DISTRIBUTIONS?
        └─ Use: Kolmogorov-Smirnov Test
           R Code: ks.test()

     KEY CONCEPTS:
     - p-value < 0.05 → SIGNIFICANT difference (reject null hypothesis)
     - p-value ≥ 0.05 → NOT significant (fail to reject null hypothesis)
     - Null hypothesis: "There is NO difference between groups"
     - Alternative hypothesis: "There IS a difference between groups"

     p-Value Interpretation Table:
     p-value | Interpretation | Decision
     --------|----------------|----------
     < 0.001 | Extremely significant | Strong evidence of difference
     < 0.01  | Highly significant | Very strong evidence
     < 0.05  | Significant | Sufficient evidence (standard threshold)
     < 0.10  | Marginally significant | Weak evidence
     ≥ 0.10  | Not significant | Insufficient evidence

     Length: 1-2 pages (visual flowchart)
     Use Case: Q4 and Q5 to quickly decide which test to use

     5. SQL_Query_Templates.sql (COPY-PASTE READY)

     Purpose: Pre-written SQL queries you can adapt for exam questions

     Contents (organized by use case):

     -- ================================================
     -- TEMPLATE 1: DAU (Daily Active Users)
     -- ================================================
     SELECT
         DATE(session_start_time) AS date,
         COUNT(DISTINCT user_id) AS DAU
     FROM sessions
     GROUP BY DATE(session_start_time)
     ORDER BY date;

     -- Adapt by: changing table name, date column name, user column name

     -- ================================================
     -- TEMPLATE 2: MAU (Monthly Active Users)
     -- ================================================
     SELECT
         DATE_FORMAT(session_start_time, '%Y-%m') AS month,
         COUNT(DISTINCT user_id) AS MAU
     FROM sessions
     GROUP BY DATE_FORMAT(session_start_time, '%Y-%m')
     ORDER BY month;

     -- ================================================
     -- TEMPLATE 3: ARPU (Average Revenue Per User)
     -- ================================================
     SELECT
         ROUND(SUM(transaction_amount) / COUNT(DISTINCT users.user_id), 2)
     AS ARPU
     FROM users
     LEFT JOIN transactions ON users.user_id = transactions.user_id;

     -- ================================================
     -- TEMPLATE 4: ARPPU (Average Revenue Per PAYING User)
     -- ================================================
     SELECT
         ROUND(SUM(transaction_amount) / COUNT(DISTINCT
     transactions.user_id), 2) AS ARPPU
     FROM transactions;

     -- Key difference: ARPU divides by ALL users, ARPPU divides by PAYING 
     users only

     -- ================================================
     -- TEMPLATE 5: Conversion Rate
     -- ================================================
     SELECT
         COUNT(DISTINCT paying_users.user_id) * 100.0 / COUNT(DISTINCT
     all_users.user_id) AS conversion_rate
     FROM
         (SELECT DISTINCT user_id FROM users) AS all_users
     LEFT JOIN
         (SELECT DISTINCT user_id FROM transactions) AS paying_users
     ON all_users.user_id = paying_users.user_id;

     -- ================================================
     -- TEMPLATE 6: D1 Retention (Complex)
     -- ================================================
     -- Step 1: Create user cohorts with signup date
     CREATE VIEW user_cohorts AS
     SELECT user_id, DATE(registration_date) AS signup_date
     FROM users;

     -- Step 2: Get all session dates per user
     CREATE VIEW user_session_dates AS
     SELECT DISTINCT user_id, DATE(session_start) AS session_date
     FROM sessions;

     -- Step 3: Check if user returned on D1
     SELECT
         uc.user_id,
         uc.signup_date,
         CASE
             WHEN usd.session_date = DATE_ADD(uc.signup_date, INTERVAL 1
     DAY) THEN 1
             ELSE 0
         END AS returned_d1
     FROM user_cohorts uc
     LEFT JOIN user_session_dates usd
         ON uc.user_id = usd.user_id
         AND usd.session_date = DATE_ADD(uc.signup_date, INTERVAL 1 DAY);

     -- Step 4: Calculate D1 retention rate
     SELECT
         COUNT(CASE WHEN returned_d1 = 1 THEN 1 END) * 100.0 / COUNT(*) AS
     d1_retention_rate
     FROM d1_return_status;

     -- ================================================
     -- TEMPLATE 7: Average Session Duration
     -- ================================================
     SELECT
         ROUND(AVG(TIMESTAMPDIFF(SECOND, start_time, end_time)), 2) AS
     avg_duration_seconds
     FROM sessions
     WHERE end_time IS NOT NULL;

     -- Convert to minutes: divide by 60
     -- Convert to hours: divide by 3600

     -- ================================================
     -- TEMPLATE 8: Session Count Per User
     -- ================================================
     SELECT
         user_id,
         COUNT(session_id) AS session_count
     FROM sessions
     GROUP BY user_id
     ORDER BY session_count DESC;

     -- For average: wrap in SELECT AVG(session_count) FROM (...)

     -- ================================================
     -- TEMPLATE 9: LEFT JOIN Pattern (CRITICAL for exam)
     -- ================================================
     SELECT
         u.user_id,
         u.username,
         COUNT(s.session_id) AS session_count
     FROM users u
     LEFT JOIN sessions s ON u.user_id = s.user_id
     GROUP BY u.user_id, u.username;

     -- LEFT JOIN keeps all users even if they have 0 sessions
     -- INNER JOIN only keeps users with at least 1 session

     -- ================================================
     -- TEMPLATE 10: Date Filtering
     -- ================================================
     SELECT *
     FROM sessions
     WHERE DATE(start_time) BETWEEN '2020-01-01' AND '2020-12-31';

     -- Or for specific month:
     WHERE DATE_FORMAT(start_time, '%Y-%m') = '2020-06';

     -- ================================================
     -- KEY SQL PATTERNS TO REMEMBER
     -- ================================================

     -- COUNT vs COUNT DISTINCT:
     COUNT(user_id)          -- Counts all rows (including duplicates)
     COUNT(DISTINCT user_id) -- Counts unique users only (use for DAU/MAU)

     -- Date functions:
     DATE(timestamp_column)                      -- Extract date only 
     (2020-01-15)
     DATE_FORMAT(timestamp_column, '%Y-%m')      -- Format as year-month 
     (2020-01)
     TIMESTAMPDIFF(SECOND, start, end)           -- Calculate difference in
      seconds
     DATE_ADD(date, INTERVAL 1 DAY)              -- Add 1 day to date

     -- Percentage calculation:
     ROUND(numerator * 100.0 / denominator, 2)   -- Always multiply by 
     100.0 (not 100)

     -- NULL handling:
     COALESCE(revenue, 0)                         -- Replace NULL with 0
     WHERE column IS NOT NULL                     -- Filter out NULLs

     -- Case when:
     CASE
         WHEN condition THEN value1
         ELSE value2
     END AS new_column_name

     # ================================================
     # SETUP: Load Required Libraries

     # ================================================

     # ================================================
     # TEMPLATE 2: Load Data from CSV (if provided)

     # Quick exploration

     str(data)            # View structure
     summary(data)        # Summary statistics
     dim(data)            # Dimensions (rows, columns)

     # ================================================
     # TEMPLATE 3: Chi-Squared Test (for proportions/categories)
     # USE WHEN: Comparing success rates, conversion rates, etc. between 
     groups
     # ================================================

     # Example: Comparing level completion rates between Group A and Group 
     B

     # Prepare data: Create contingency table
     # Format:
     #           Success  Failure
     # Group A      X        Y
     # Group B      Z        W

     group_a_success <- 250
     group_a_failure <- 750
     group_b_success <- 180
     group_b_failure <- 820

     # Create matrix
     contingency_table <- matrix(
         c(group_a_success, group_a_failure,
           group_b_success, group_b_failure),
         nrow = 2,
         byrow = TRUE
     )
     rownames(contingency_table) <- c("Group_A", "Group_B")
     colnames(contingency_table) <- c("Success", "Failure")

     # Print table to verify
     print(contingency_table)

     # Run chi-squared test
     chi_result <- chisq.test(contingency_table)

     # View results
     print(chi_result)

     # Extract key values
     p_value <- chi_result$p.value
     chi_statistic <- chi_result$statistic

     # Interpretation
     if (p_value < 0.05) {
         cat("SIGNIFICANT: p-value =", p_value, "\n")
         cat("Reject null hypothesis: There IS a significant difference 
     between groups\n")
     } else {
         cat("NOT SIGNIFICANT: p-value =", p_value, "\n")
         cat("Fail to reject null hypothesis: No significant difference 
     between groups\n")
     }

     # Calculate success rates for context
     group_a_rate <- group_a_success / (group_a_success + group_a_failure)
     group_b_rate <- group_b_success / (group_b_success + group_b_failure)
     cat("Group A success rate:", round(group_a_rate * 100, 2), "%\n")
     cat("Group B success rate:", round(group_b_rate * 100, 2), "%\n")
     cat("Difference:", round((group_a_rate - group_b_rate) * 100, 2),
     "percentage points\n")

     # ================================================
     # TEMPLATE 4: Proportion Test (alternative to Chi-Squared)
     # USE WHEN: Directly comparing two proportions
     # ================================================

     # Example: Same scenario as above
     prop_result <- prop.test(
         x = c(group_a_success, group_b_success),  # Number of successes
         n = c(group_a_success + group_a_failure,
               group_b_success + group_b_failure)  # Total trials
     )

     print(prop_result)

     # Confidence interval for difference
     conf_int <- prop_result$conf.int
     cat("95% CI for difference:", conf_int[1], "to", conf_int[2], "\n")

     # ================================================
     # TEMPLATE 5: t-Test (for comparing means)
     # USE WHEN: Comparing average session length, average revenue, etc.
     # ================================================

     # Example: Comparing average session length between two groups

     # Prepare data
     group_a_sessions <- c(20, 25, 30, 22, 28, 35, 18, 24)  # Example data
     group_b_sessions <- c(15, 18, 20, 17, 16, 22, 19, 21)  # Example data

     # group_a_sessions <- data %>% filter(group == "A") %>%

     # group_b_sessions <- data %>% filter(group == "B") %>% 
     pull(session_length)

     # Run independent t-test
     t_result <- t.test(group_a_sessions, group_b_sessions)

     # View results
     print(t_result)

     # Extract key values
     p_value <- t_result$p.value
     mean_a <- t_result$estimate[1]
     mean_b <- t_result$estimate[2]
     conf_int <- t_result$conf.int

     # Interpretation
     if (p_value < 0.05) {
         cat("SIGNIFICANT: p-value =", p_value, "\n")
         cat("The difference in means is statistically significant\n")
     } else {
         cat("NOT SIGNIFICANT: p-value =", p_value, "\n")
         cat("The difference in means is not statistically significant\n")

     cat("Group B mean:", round(mean_b, 2), "\n")

     7. AB_Testing_Complete_Toolkit.md (CRITICAL - FOR Q5 WORTH 4 POINTS)

     Purpose: Complete framework for analyzing A/B tests and making
     business decisions

     Contents:

     A/B Testing Analysis Workflow

     STEP 1: Understand the Problem (2 minutes)
     Checklist:
     [ ] What is being tested? (feature change, difficulty adjustment, UI
     change, etc.)
     [ ] What are the groups? (Control = B, Treatment = A, or vice versa)
     [ ] What is the primary metric? (success rate, session length,
     revenue, etc.)
     [ ] What is the hypothesis? (A will be better/worse than B)
     [ ] What is the business goal? (increase retention, increase revenue,
     improve UX)

     STEP 2: Load and Explore Data (5 minutes)
     # Load data from provided file or database
     data <- read.csv("sweet.sql", header = TRUE)  # Or from MySQL

     # Quick exploration
     head(data)
     str(data)
     table(data$group)  # Check group sizes

     # Verify data quality
     summary(data)
     sum(is.na(data))  # Check for missing values

     STEP 3: Calculate Group Metrics (10 minutes)
     # For PROPORTIONS (success rate, completion rate, etc.)
     library(dplyr)

     group_summary <- data %>%
         group_by(group) %>%
         summarize(
             total_attempts = n(),
             successes = sum(success == 1),  # Adjust based on data
             failures = sum(success == 0),
             success_rate = mean(success) * 100
         )

     print(group_summary)

     # For MEANS (session length, revenue, etc.)
     group_means <- data %>%
         group_by(group) %>%
         summarize(
             count = n(),
             mean_value = mean(metric, na.rm = TRUE),
             sd_value = sd(metric, na.rm = TRUE)
         )

     print(group_means)

     STEP 4: Run Statistical Test (10 minutes)
     # DECISION: Use decision tree to choose test
     # - Comparing proportions/rates? → Chi-squared or prop.test
     # - Comparing means/averages? → t-test

     # OPTION A: Chi-Squared Test (most common for A/B tests)
     group_a <- group_summary %>% filter(group == "A")
     group_b <- group_summary %>% filter(group == "B")

     contingency_table <- matrix(
         c(group_a$successes, group_a$failures,
           group_b$successes, group_b$failures),
         nrow = 2,
         byrow = TRUE
     )

     chi_result <- chisq.test(contingency_table)
     p_value <- chi_result$p.value

     print(chi_result)

     # OPTION B: t-test (if comparing means)
     t_result <- t.test(metric ~ group, data = data)
     p_value <- t_result$p.value

     print(t_result)

     STEP 5: Interpret Results (5 minutes)
     # Statistical Significance
     if (p_value < 0.05) {
         cat("RESULT: Statistically significant difference (p =",
     round(p_value, 4), ")\n")
         cat("INTERPRETATION: The difference is unlikely due to chance\n")
     } else {
         cat("RESULT: NOT statistically significant (p =", round(p_value,
     4), ")\n")
         cat("INTERPRETATION: The difference could be due to random 
     chance\n")
     }

     # Practical Significance (Effect Size)
     difference <- group_a$success_rate - group_b$success_rate
     cat("Effect size:", round(difference, 2), "percentage points\n")

     # Business Impact
     # Calculate potential impact if rolled out to all users
     # Example: 1 million players, 2.5% improvement = 25,000 more successes

     STEP 6: Make Business Recommendation (8 minutes)

     Decision Framework:
     IF p-value < 0.05 AND effect is positive (desirable):
         → RECOMMEND: Implement the change for all users
         → REASON: Statistically significant improvement

     IF p-value < 0.05 AND effect is negative (undesirable):
         → RECOMMEND: DO NOT implement, revert to control
         → REASON: Change makes things worse

     IF p-value ≥ 0.05:
         → RECOMMEND: Need more data OR no clear winner
         → REASON: Difference not statistically significant

     Answer Template for Q5:
     **Recommendation**: [Implement / Do Not Implement / Need More Data]

     **Statistical Evidence**:
     - Group A [metric]: X% (n = N)
